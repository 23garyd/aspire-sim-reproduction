"""B601 cube-lift task. FakeCubeLiftEnv is a deterministic Cartesian model (no Isaac,
no physics): a correct reach->grasp->lift program succeeds, a no-op or wrong-pose
program fails, so the reward is meaningful and the whole engine contract is testable
without Isaac. IsaacCubeLiftEnv (below) is the live, on-box counterpart.

Simplification (honest): grasp = gripper closes within grasp_tol of the cube; while
grasped the cube follows the EE; success = final cube height > lift_threshold. Real
grasp physics is Isaac-only."""
from __future__ import annotations

import random
from dataclasses import dataclass

from aspire.schemas import TraceRecord


@dataclass
class CubePose:
    x: float
    y: float
    z: float
    qx: float = 0.0
    qy: float = 0.0
    qz: float = 0.0
    qw: float = 1.0


class FakeCubeLiftEnv:
    def __init__(self, grasp_tol: float = 0.03, lift_threshold: float = 0.10,
                 table_z: float = 0.0, cube_xy=(0.4, 0.0)):
        self.grasp_tol = grasp_tol
        self.lift_threshold = lift_threshold
        self.table_z = table_z
        self._cube_xy = cube_xy
        self.reset(0)

    def reset(self, seed: int) -> None:
        rng = random.Random(seed)                      # seed-deterministic cube pose
        dx = (rng.random() - 0.5) * 0.10
        dy = (rng.random() - 0.5) * 0.10
        self.cube = CubePose(self._cube_xy[0] + dx, self._cube_xy[1] + dy, self.table_z)
        self.ee = [0.0, 0.0, 0.30]
        self.grip = 0.04
        self.grasped = False
        self._traces: list[TraceRecord] = []

    def _trace(self, function: str, args: dict, summary: str) -> None:
        self._traces.append(TraceRecord(step=len(self._traces), function=function,
                                        args=args, status="ok", result_summary=summary,
                                        duration_ms=0.0, media=[]))

    def get_cube_pose(self) -> CubePose:
        self._trace("get_cube_pose", {}, f"cube=({self.cube.x:.3f},{self.cube.y:.3f},{self.cube.z:.3f})")
        return self.cube

    def move_to_pose(self, x, y, z, roll=0.0, pitch=0.0, yaw=0.0) -> str:
        self.ee = [float(x), float(y), float(z)]
        if self.grasped:
            self.cube = CubePose(float(x), float(y), float(z))   # cube rides the EE
        self._trace("move_to_pose", {"x": x, "y": y, "z": z}, f"ee=({x:.3f},{y:.3f},{z:.3f})")
        return "ok"

    def set_gripper(self, pos: float) -> str:
        self.grip = float(pos)
        if pos < 0.02 and not self.grasped:            # closing
            d = ((self.ee[0] - self.cube.x) ** 2 + (self.ee[1] - self.cube.y) ** 2
                 + (self.ee[2] - self.cube.z) ** 2) ** 0.5
            if d <= self.grasp_tol:
                self.grasped = True
        self._trace("set_gripper", {"pos": pos}, f"grip={pos} grasped={self.grasped}")
        return "ok"

    def open_gripper(self) -> str:
        return self.set_gripper(0.04)

    def close_gripper(self) -> str:
        return self.set_gripper(0.0)

    def namespace(self) -> dict:
        return {"get_cube_pose": self.get_cube_pose, "move_to_pose": self.move_to_pose,
                "set_gripper": self.set_gripper, "open_gripper": self.open_gripper,
                "close_gripper": self.close_gripper}

    def success(self) -> bool:
        return self.grasped and self.cube.z > self.lift_threshold

    def traces(self) -> list:
        return list(self._traces)
```

#!/usr/bin/env python3
"""B601 cube-lift engine. Honors the capx_execute contract so it is a drop-in for
`aspire.run --engine capx --capx-cmd`. Per seed: reset env, audit the program text
(reused forbidden-API wall), exec it against the env's privileged primitive namespace,
score by env.success(). --selftest uses the deterministic FakeCubeLiftEnv (no Isaac);
a real --config-path uses IsaacCubeLiftEnv (on-box)."""
import argparse
import json
import os
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[1]))   # repo root importable

from aspire.audit import audit_code
from robot_b601.task import FakeCubeLiftEnv


def _err_trace(step, function, summary):
    return {"step": step, "function": function, "args": {}, "status": "error",
            "result_summary": summary, "duration_ms": 0.0, "media": []}


def run_episode(program_path, seed, env):
    env.reset(seed)
    src = Path(program_path).read_text()
    violations = audit_code(src)
    if violations:
        return False, [_err_trace(0, "audit", "; ".join(violations))]
    ns = env.namespace()
    try:
        exec(src, ns)                       # program calls primitives by bare name
        ok = env.success()
    except Exception as e:                  # a raising program fails the seed, not the run
        traces = [t.to_dict() for t in env.traces()]
        traces.append(_err_trace(len(traces), "program_error", f"{type(e).__name__}: {e}"))
        return False, traces
    return ok, [t.to_dict() for t in env.traces()]


def _build_env(args):
    if args.selftest or not args.config_path:
        return FakeCubeLiftEnv()
    from robot_b601.task import IsaacCubeLiftEnv     # deferred: Isaac only on the live path
    return IsaacCubeLiftEnv(args.config_path)


def main(argv=None):
    ap = argparse.ArgumentParser()
    ap.add_argument("--program", required=True)
    ap.add_argument("--seeds", required=True)
    ap.add_argument("--out-dir", required=True)
    ap.add_argument("--config-path", default="")
    ap.add_argument("--selftest", action="store_true")
    ap.add_argument("--workdir", default="")
    args = ap.parse_args(argv)
    if args.workdir:
        os.chdir(args.workdir)

    out_dir = Path(args.out_dir)
    out_dir.mkdir(parents=True, exist_ok=True)
    seeds = [int(s) for s in args.seeds.split(",") if s.strip()]
    env = _build_env(args)

    successes, all_traces = 0, []
    for seed in seeds:
        ok, traces = run_episode(args.program, seed, env)
        successes += bool(ok)
        all_traces.extend(traces)

    tmp = out_dir / "result.json.tmp"
    tmp.write_text(json.dumps(
        {"score": successes / len(seeds), "raw_dir": str(out_dir),
         "traces": all_traces}, indent=1))
    os.replace(tmp, out_dir / "result.json")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
